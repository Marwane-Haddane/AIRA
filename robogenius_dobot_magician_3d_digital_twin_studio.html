dobot_blender_generator.py (Blender 3.x / 4.x Python API) Copied to clipboard! 
# Blender 4.x Procedural Generator: Dobot Magician Classic 4-Axis Digital Twin
# Run inside Blender -> Scripting Workspace -> New -> Run Script
import bpy, math

bpy.ops.wm.read_factory_settings(use_empty=True)

def make_mat(name, color, metal=0.0, rough=0.25, clearcoat=0.0):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Metallic"].default_value = metal
    bsdf.inputs["Roughness"].default_value = rough
    if "Coat Weight" in bsdf.inputs:
        bsdf.inputs["Coat Weight"].default_value = clearcoat
    return mat

mat_white = make_mat("Dobot_White_ABS", (0.95, 0.96, 0.98, 1), 0.05, 0.22, 0.8)
mat_black = make_mat("Dobot_Black_Anodized", (0.08, 0.09, 0.11, 1), 0.75, 0.35, 0.0)
mat_steel = make_mat("Dobot_Steel_Chrome", (0.82, 0.85, 0.88, 1), 0.92, 0.15, 0.0)

# 1. Stationary Base Pedestal (158 x 158 x 50 mm)
bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0.025))
base = bpy.context.active_object
base.name = "01_Dobot_Base"
base.scale = (0.158, 0.158, 0.050)
bpy.ops.object.modifier_add(type='BEVEL')
base.modifiers["Bevel"].width = 0.012
base.modifiers["Bevel"].segments = 5
base.data.materials.append(mat_white)

# 2. J1 Rotating Turret & Dual NEMA 17 Stepper Motors
bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0, 0, 0.05))
j1_pivot = bpy.context.active_object
j1_pivot.name = "J1_Turret_Yaw_Pivot"
j1_pivot.parent = base

bpy.ops.mesh.primitive_cube_add(size=1, location=(0, 0, 0.045))
turret = bpy.context.active_object
turret.name = "02_Turret_Housing"
turret.scale = (0.088, 0.084, 0.090)
turret.parent = j1_pivot
turret.data.materials.append(mat_white)

for side in [-1, 1]:
    bpy.ops.mesh.primitive_cube_add(size=1, location=(0, side * 0.064, 0.058))
    motor = bpy.context.active_object
    motor.name = f"NEMA17_Stepper_{'L' if side<0 else 'R'}"
    motor.scale = (0.042, 0.046, 0.042)
    motor.parent = j1_pivot
    motor.data.materials.append(mat_black)

# 3. J2 Rear Arm (135 mm) + White Upper Armor Shroud
bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0, 0, 0.075))
j2_pivot = bpy.context.active_object
j2_pivot.name = "J2_Shoulder_Pitch_Pivot"
j2_pivot.parent = j1_pivot

bpy.ops.mesh.primitive_cube_add(size=1, location=(0.0675, 0, 0))
rear_arm = bpy.context.active_object
rear_arm.name = "03_Rear_Arm_Linkage"
rear_arm.scale = (0.135, 0.052, 0.032)
rear_arm.parent = j2_pivot
rear_arm.data.materials.append(mat_black)

# 4. J3 Forearm (147 mm) & Parallel Linkage Wrist
bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0.135, 0, 0))
j3_pivot = bpy.context.active_object
j3_pivot.name = "J3_Elbow_Pitch_Pivot"
j3_pivot.parent = j2_pivot

bpy.ops.mesh.primitive_cube_add(size=1, location=(0.0735, 0, 0))
forearm = bpy.context.active_object
forearm.name = "04_Forearm_Linkage"
forearm.scale = (0.147, 0.044, 0.028)
forearm.parent = j3_pivot
forearm.data.materials.append(mat_black)

# 5. J4 Precision Pen Holder End-Effector
bpy.ops.mesh.primitive_cylinder_add(radius=0.008, depth=0.135, location=(0.175, 0, -0.01))
pen = bpy.context.active_object
pen.name = "05_Pen_Tool_EndEffector"
pen.parent = j3_pivot
pen.data.materials.append(mat_steel)

print("Dobot Magician Classic 3D Hierarchy Created Successfully!")
